def is_prime(n):
    """Helper function to check if a number is prime."""
    if n <= 1:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    return True

def primes():
    """Generator function to yield prime numbers one at a time."""
    n = 2
    while n <= 100:
        if is_prime(n):
            yield n
        n += 1

# Using the primes() generator
for prime in primes():
    print(prime)
